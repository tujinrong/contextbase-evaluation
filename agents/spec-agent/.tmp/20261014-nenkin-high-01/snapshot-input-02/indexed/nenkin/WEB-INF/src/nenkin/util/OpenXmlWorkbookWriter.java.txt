000270 |     private static String validXmlText(String value) {
000271 |         StringBuilder result = new StringBuilder(value == null ? 0 : value.length());
000272 |         String source = value == null ? "" : value;
000273 |         for (int offset = 0; offset < source.length();) {
000274 |             int codePoint = source.codePointAt(offset);
000275 |             if (codePoint == 0x9 || codePoint == 0xA || codePoint == 0xD
000276 |                     || (codePoint >= 0x20 && codePoint <= 0xD7FF)
000277 |                     || (codePoint >= 0xE000 && codePoint <= 0xFFFD)
000278 |                     || (codePoint >= 0x10000 && codePoint <= 0x10FFFF)) {
000279 |                 result.appendCodePoint(codePoint);
000280 |             }
000281 |             offset += Character.charCount(codePoint);
000282 |         }
000283 |         return result.toString();
000284 |     }
