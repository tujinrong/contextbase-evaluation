000034 |     /** yyyy-MM-dd、yyyy/MM/dd、yyyyMMdd、和暦を yyyy-MM-dd に正規化する。 */
000035 |     public static String toIsoDate(String value) {
000036 |         String input = normalizeInput(value);
000037 |         if (input.length() == 0) {
000038 |             return "";
000039 |         }
000040 | 
000041 |         Matcher eraMatcher = ERA_DATE.matcher(input);
000042 |         if (eraMatcher.matches()) {
000043 |             Era era = findEra(eraMatcher.group(1));
000044 |             int eraYear = "元".equals(eraMatcher.group(2))
000045 |                     ? 1 : Integer.parseInt(eraMatcher.group(2));
000046 |             if (eraYear <= 0) {
000047 |                 throw invalid(value);
000048 |             }
000049 |             int year = era.baseYear + eraYear - 1;
000050 |             String iso = formatIso(year, Integer.parseInt(eraMatcher.group(3)),
000051 |                     Integer.parseInt(eraMatcher.group(4)), value);
000052 |             Date date = parseIso(iso, value);
000053 |             if (date.before(era.start) || (era.end != null && date.after(era.end))) {
000054 |                 throw invalid(value);
000055 |             }
000056 |             return iso;
000057 |         }
000058 | 
000059 |         Matcher western = GREGORIAN_DATE.matcher(input);
000060 |         if (western.matches()) {
000061 |             return formatIso(Integer.parseInt(western.group(1)),
000062 |                     Integer.parseInt(western.group(2)), Integer.parseInt(western.group(3)), value);
000063 |         }
000064 | 
000065 |         Matcher compact = COMPACT_DATE.matcher(input);
000066 |         if (compact.matches()) {
000067 |             return formatIso(Integer.parseInt(compact.group(1)),
000068 |                     Integer.parseInt(compact.group(2)), Integer.parseInt(compact.group(3)), value);
