000058 |     public PensionBatchExportResult build(PensionBatchExportRequest request) {
000059 |         if (request == null) {
000060 |             throw new IllegalArgumentException("出力条件がありません。");
000061 |         }
000062 |         if (!request.isIncludeNew() && !request.isIncludeChanged()
000063 |                 && !request.isIncludeDeleted()) {
000064 |             throw new IllegalArgumentException(
000065 |                     "新規分、変更分、削除分のいずれかを選択してください。");
000066 |         }
000067 | 
000068 |         DateRange range = resolveRange(request);
000069 |         Integer updatedBy = parseStaffNo(request.getStaffNo());
000070 |         if (request.getDeleteCutoffDate().trim().length() > 0) {
000071 |             repository.purgeDeletedAuditsThrough(request.getDeleteCutoffDate());
000072 |         }
000073 | 
000074 |         List<ExportSpec> specs = selectedSpecs(request, updatedBy != null);
000075 |         List<GeneratedFile> generated = new ArrayList<GeneratedFile>();
000076 |         List<String> messages = new ArrayList<String>();
000077 |         for (int i = 0; i < specs.size(); i++) {
000078 |             ExportSpec spec = specs.get(i);
000079 |             List<PensionAuditRecord> audits = repository.findAuditRecords(spec.type,
000080 |                     range.from, range.to, updatedBy);
000081 |             if (audits.isEmpty()) {
000082 |                 messages.add(emptyMessage(spec.type, request.getRunMode(), updatedBy != null));
000083 |                 continue;
000084 |             }
000085 |             sortAudits(audits);
000086 |             List<String> headers = spec.type == AuditType.NEW
000087 |                     ? NEW_HEADERS : CHANGE_DELETE_HEADERS;
000088 |             List<Object[]> rows = toRows(audits, spec.type);
000089 |             WorkbookLayout layout = buildLayout(spec, request, range, headers.size());
000090 |             byte[] content = OpenXmlWorkbookWriter.buildWorkbook(spec.sheetName, headers, rows,
000091 |                     layout.leadingRows, layout.mergedRanges, layout.printScale);
000092 |             generated.add(new GeneratedFile(spec, rows.size(), content));
000093 |         }
000094 | 
000095 |         List<PensionBatchExportFileResult> files = toFileResults(generated);
000096 |         PensionBatchExportResult result;
000097 |         if (generated.isEmpty()) {
000098 |             result = new PensionBatchExportResult("", EXCEL_CONTENT_TYPE, new byte[0],
000099 |                     files, messages);
000100 |         } else if (generated.size() == 1) {
000101 |             GeneratedFile file = generated.get(0);
000102 |             result = new PensionBatchExportResult(file.spec.fileName, EXCEL_CONTENT_TYPE,
000103 |                     file.content, files, messages);

000105 |             String zipName = request.getRunMode() == PensionBatchRunMode.DAILY
000106 |                     ? "年金予約データ明細リスト（日次）.zip"

000108 |             result = new PensionBatchExportResult(zipName, ZIP_CONTENT_TYPE,
000109 |                     buildZip(generated, messages), files, messages);

000115 |     private static DateRange resolveRange(PensionBatchExportRequest request) {

000313 |     private static byte[] buildZip(List<GeneratedFile> files, List<String> messages) {
000314 |         ByteArrayOutputStream output = new ByteArrayOutputStream();

000318 |                 GeneratedFile file = files.get(i);
000319 |                 writeZipEntry(zip, file.spec.fileName, file.content);
000320 |             }
000321 |             if (!messages.isEmpty()) {
000322 |                 StringBuilder text = new StringBuilder();
000323 |                 for (int i = 0; i < messages.size(); i++) {
000324 |                     text.append(messages.get(i)).append("\r\n");
000325 |                 }
000326 |                 writeZipEntry(zip, "出力結果.txt", text.toString().getBytes(UTF8));
000327 |             }
000328 |             zip.finish();
000329 |             return output.toByteArray();
000330 |         } catch (IOException e) {
000331 |             throw new IllegalStateException("ZIPファイルの生成に失敗しました。", e);
000332 |         } finally {
000333 |             try {
000334 |                 zip.close();

000336 |                 // ByteArrayOutputStream では close 失敗しない。
