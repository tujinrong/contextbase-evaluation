Option Explicit
Option Compare Binary

Public Function LoadStudent(ByVal id As Long) As Object
    Dim tx As New clsUnitOfWork, repo As New clsStudentRepository
    Authorize "Read"
    On Error GoTo Failed
    tx.BeginWork: Set repo.Database = tx.Database
    Set LoadStudent = repo.LoadStudent(id)
    tx.CloseWork
    If LoadStudent Is Nothing Then Err.Raise vbObjectError + 127, , "E-STU-103"
    Exit Function
Failed:
    Dim n As Long, s As String
    n = Err.number: s = Err.description
    On Error Resume Next: tx.CloseWork: On Error GoTo 0
    Err.Raise n, , s
End Function

Public Function SaveStudent(ByVal raw As Object, ByVal target As Variant, ByVal expected As Variant, ByVal formId As String, ByVal confirmation As clsConfirmation, ByVal confirmed As Boolean, ByVal reasonRaw As Variant) As Object
    Dim tx As New clsUnitOfWork, repo As New clsStudentRepository, d As Object, old As Object, result As Object
    Dim utc As Date, day As String, user As String, id As Long, version As Long, before As Variant, reason As Variant
    Dim candidates As Collection, digest As String, changed As String, k As Variant, same As Boolean, warning As String
    Set result = Map(): Set SaveStudent = result
    On Error GoTo Failed
    Authorize "Save": utc = UtcNow(): day = businessDay(utc): user = CurrentUserId()
    Set d = NormalizeDraft(raw, day): reason = TextValue(reasonRaw, 60, 60, "memo")
    If Not IsNull(target) Then
        id = PositiveId(target): expected = PositiveId(expected)
    ElseIf Not IsNull(expected) Then
        Err.Raise vbObjectError + 119, , "E-COM-003"
    End If
    tx.BeginWork: Set repo.Database = tx.Database
    If id <> 0 Then
        Set old = repo.LoadStudent(id)
        If old Is Nothing Then Err.Raise vbObjectError + 127, , "E-STU-103"
        If old("row_version") <> expected Then Err.Raise vbObjectError + 126, , "E-STU-105"
        If old("student_no") <> d("student_no") Then Err.Raise vbObjectError + 128, , "E-STU-104"
        before = old("row_version")
        If (old("enrollment_status_code") = "03" Or old("enrollment_status_code") = "04") And old("enrollment_status_code") <> d("enrollment_status_code") Then
            Authorize "CorrectEnded"
            If IsNull(reason) Then Err.Raise vbObjectError + 129, , "E-STU-106"
        End If
    Else
        before = Null
    End If
    repo.CheckUnique d("student_no"), id
    CheckMasters repo, d, old, raw, day, id, warning
    same = Not old Is Nothing
    If same Then
        For Each k In d.Keys
            If Not EqualValue(d(k), old(k)) Then same = False
        Next
    End If
    If same Then
        tx.CloseWork: result.Add "kind", "NoChange": result.Add "message", "I-STU-004": result.Add "warning", warning: Exit Function
    End If
    Set candidates = repo.Duplicates(d, id, digest)
    If candidates.count > 0 Or confirmation.Token <> "" Then
        If Not confirmed Or Not confirmation.Valid(formId, target, expected, d, digest) Then
            confirmation.Issue formId, target, expected, d, digest
            tx.CloseWork
            result.Add "kind", "ConfirmationRequired": result.Add "message", IIf(confirmed, "W-STU-102", "W-STU-101")
            result.Add "candidates", candidates: Exit Function
        End If
        If IsNull(reason) Then Err.Raise vbObjectError + 129, , "E-STU-107"
    End If
    id = repo.WriteStudent(d, old, utc, user, version, changed)
    repo.AppendAudit id, IIf(old Is Nothing, "CREATE", "UPDATE"), before, version, changed, reason, candidates.count > 0, utc, user
    tx.Commit
    confirmation.Consumed = True
    tx.CloseWork
    result.Add "kind", "Saved": result.Add "student_id", id: result.Add "row_version", version
    result.Add "student_no", d("student_no"): result.Add "warning", warning
    result.Add "message", IIf(old Is Nothing, "I-STU-001", "I-STU-002")
    Exit Function
Failed:
    Dim e As String, unknown As Boolean, errorNumber As Long
    e = Err.description: errorNumber = Err.number: unknown = tx.CommitStarted
    e = FailureMessage(e, errorNumber, "SaveStudent")
    result.Add "diagnostic", "Error " & CStr(errorNumber)
    On Error Resume Next: tx.CloseWork: On Error GoTo 0
    If Left$(e, 2) <> "E-" Then e = "E-COM-004"
    result.Add "kind", IIf(unknown, "Unknown", ErrorKind(e))
    result.Add "message", e
    result.Add "errorCode", errorNumber
End Function

Private Sub CheckMasters(ByVal repo As clsStudentRepository, ByVal d As Object, ByVal old As Object, ByVal raw As Object, ByVal day As String, ByVal id As Long, ByRef warning As String)
    Dim pair As Variant, oldValue As Variant, c As Object, oldSeat As Variant, newSeat As Variant, changedPeriod As Boolean
    For Each pair In Array(Array("GENDER", "gender_code"), Array("PREFECTURE", "prefecture_code"), Array("ENROLLMENT_STATUS", "enrollment_status_code"), Array("RELATIONSHIP", "relationship_code"))
        oldValue = Null
        If Not old Is Nothing Then oldValue = old(pair(1))
        repo.CheckCode pair(0), d(pair(1)), oldValue, day
    Next
    If raw.Exists("school_year") Then
        If Nz(raw("school_year"), "") <> "" Then
            If Not Match(NFKC(CStr(raw("school_year"))), "^[0-9]{4}$") Then Err.Raise vbObjectError + 130, , "E-DATE-003|field=年度"
            If CLng(NFKC(raw("school_year"))) < 1900 Or CLng(NFKC(raw("school_year"))) > 9998 Then Err.Raise vbObjectError + 130, , "E-DATE-003|field=年度"
        End If
    End If
    If raw.Exists("grade_code") Then
        If Nz(raw("grade_code"), "") <> "" Then
            If Not Match(NFKC(CStr(raw("grade_code"))), "^[0-9]{2}$") Then Err.Raise vbObjectError + 130, , "E-STU-009|field=学年"
        End If
    End If
    If IsNull(d("class_id")) Then Exit Sub
    Set c = repo.ClassById(d("class_id"))
    If c Is Nothing Then Err.Raise vbObjectError + 130, , "E-STU-010"
    If raw.Exists("school_year") Then
        If Nz(raw("school_year"), "") <> "" Then
            If Not Match(NFKC(CStr(raw("school_year"))), "^[0-9]{4}$") Then Err.Raise vbObjectError + 130, , "E-DATE-003"
            If CLng(NFKC(raw("school_year"))) < 1900 Or CLng(NFKC(raw("school_year"))) > 9998 Or CLng(NFKC(raw("school_year"))) <> c("school_year") Then Err.Raise vbObjectError + 130, , "E-STU-014"
        End If
    End If
    If raw.Exists("grade_code") Then
        If Nz(raw("grade_code"), "") <> "" Then
            If NFKC(raw("grade_code")) <> c("grade_code") Then Err.Raise vbObjectError + 130, , "E-STU-014"
        End If
    End If
    oldSeat = Seat(old): newSeat = Seat(d)
    If Not IsNull(newSeat) Then
        If Not EqualValue(oldSeat, newSeat) Then
            If Not c("is_active") Or day < c("valid_from") Or day > c("valid_to") Then Err.Raise vbObjectError + 130, , "E-STU-013"
            If Not IsNull(c("capacity")) Then
                If repo.Occupancy(d("class_id"), id) + 1 > c("capacity") Then Err.Raise vbObjectError + 130, , "E-STU-102"
            End If
        ElseIf Not c("is_active") Or day < c("valid_from") Or day > c("valid_to") Then
            warning = "W-STU-106"
        End If
    Else
        changedPeriod = old Is Nothing
        If Not old Is Nothing Then changedPeriod = Not EqualValue(old("class_id"), d("class_id")) Or Not EqualValue(old("admission_date"), d("admission_date")) Or Not EqualValue(old("enrollment_end_date"), d("enrollment_end_date")) Or Not EqualValue(old("enrollment_status_code"), d("enrollment_status_code"))
        If changedPeriod Then
            If c("valid_from") > d("enrollment_end_date") Or c("valid_to") < d("admission_date") Then Err.Raise vbObjectError + 130, , "E-STU-108"
        End If
    End If
End Sub

Public Function DeleteStudent(ByVal target As Variant, ByVal expected As Variant, ByVal reasonRaw As Variant, ByVal explicitConfirm As Boolean) As Object
    Dim tx As New clsUnitOfWork, repo As New clsStudentRepository, old As Object, result As Object
    Dim id As Long, version As Long, reason As Variant, utc As Date, user As String
    Set result = Map(): Set DeleteStudent = result
    On Error GoTo Failed
    Authorize "Delete"
    If Not explicitConfirm Then Err.Raise vbObjectError + 131, , "E-STU-020"
    id = PositiveId(target): version = PositiveId(expected)
    reason = TextValue(reasonRaw, 60, 60, "memo")
    If IsNull(reason) Then Err.Raise vbObjectError + 131, , "E-STU-020"
    utc = UtcNow(): user = CurrentUserId()
    tx.BeginWork: Set repo.Database = tx.Database: Set old = repo.LoadStudent(id)
    If old Is Nothing Then Err.Raise vbObjectError + 127, , "E-STU-103"
    If old("row_version") <> version Then Err.Raise vbObjectError + 126, , "E-STU-105"
    repo.DeleteStudent old, reason, utc, user
    repo.AppendAudit id, "DELETE", version, version + 1, "[""is_deleted"",""deleted_at"",""deleted_by"",""delete_reason"",""updated_at"",""updated_by"",""row_version""]", reason, False, utc, user
    tx.Commit: tx.CloseWork
    result.Add "kind", "Saved": result.Add "message", "I-STU-003"
    Exit Function
Failed:
    Dim e As String, unknown As Boolean, errorNumber As Long
    e = Err.description: errorNumber = Err.number: unknown = tx.CommitStarted
    e = FailureMessage(e, errorNumber, "DeleteStudent")
    On Error Resume Next: tx.CloseWork: On Error GoTo 0
    If Left$(e, 2) <> "E-" Then e = "E-COM-004"
    result.Add "kind", IIf(unknown, "Unknown", ErrorKind(e))
    result.Add "message", e
    result.Add "errorCode", errorNumber
End Function
